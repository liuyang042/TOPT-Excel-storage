' Execute expiration check before any protected operation
Public Function CheckExpiration() As Boolean
    On Error GoTo ErrorHandler
    
    Dim githubLink As String
    Dim http As Object
    Dim expireDate As Date
    Dim today As Date
    Dim responseText As String
    
    ' GitHub expiration file link
    githubLink = "https://raw.githubusercontent.com/liuyang042/ExcelLicense/refs/heads/main/expire.txt"
    
    ' Create and send HTTP request
    Set http = CreateObject("MSXML2.XMLHTTP.6.0")
    http.Open "GET", githubLink, False
    http.setRequestHeader "Content-Type", "text/plain"
    http.setRequestHeader "User-Agent", "Excel-Expiration-Check"
    http.Send
    
    ' Check HTTP status - must be 200 (success)
    If http.Status <> 200 Then
        DisplayErrorMessage "Failed to connect to license server. Please check your internet connection."
        CheckExpiration = False
        Exit Function
    End If
    
    ' Parse and validate response
    responseText = Trim(Replace(Replace(http.ResponseText, vbLf, ""), vbCr, ""))
    
    If Not IsDate(responseText) Then
        DisplayErrorMessage "Invalid expiration date format received from server."
        CheckExpiration = False
        Exit Function
    End If
    
    ' Check expiration
    expireDate = DateValue(responseText)
    today = Date
    
    If today > expireDate Then
        DisplayExpirationMessage expireDate, today
        CheckExpiration = False
    Else
        CheckExpiration = True
    End If
    
    Exit Function

ErrorHandler:
    DisplayErrorMessage "Error during license verification: " & Err.Description
    CheckExpiration = False
End Function

' Display error message
Private Sub DisplayErrorMessage(ByVal message As String)
    Dim fullMessage As String
    fullMessage = message & vbCrLf & vbCrLf & _
                  "Please contact author: Liu Yang" & vbCrLf & _
                  "Email: luckilyliuyang@163.com"
    MsgBox fullMessage, vbCritical, "License Verification Failed"
End Sub

' Display expiration message
Private Sub DisplayExpirationMessage(ByVal expDate As Date, ByVal currDate As Date)
    Dim message As String
    message = "Version expired." & vbCrLf & vbCrLf & _
              "Expiration Date: " & Format(expDate, "yyyy-mm-dd") & vbCrLf & _
              "Current Date: " & Format(currDate, "yyyy-mm-dd") & vbCrLf & vbCrLf & _
              "Please contact author for updated version:" & vbCrLf & _
              "Liu Yang" & vbCrLf & _
              "Email: luckilyliuyang@163.com"
    MsgBox message, vbCritical, "Version Expired"
End Sub

' =============================================
' Worksheet Activate Event with Protection
' Place this in the DATA sheet module
' =============================================
Private Sub Worksheet_Activate()
    ' Execute expiration verification first
    If Not CheckExpiration() Then
        ' If expired, deactivate this sheet
        Application.EnableEvents = False
        If ThisWorkbook.Sheets.Count > 1 Then
            ThisWorkbook.Sheets(1).Activate
        End If
        Application.EnableEvents = True
        Exit Sub
    End If
    
    ' Continue with normal activation if license valid
    Range("A2").Select
End Sub
