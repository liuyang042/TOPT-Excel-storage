' Universal Expiration Check Module

' Execute expiration check before any protected operation
Public Function CheckExpiration() As Boolean
    On Error GoTo ErrorHandler
    
    Dim filePath As String
    Dim fileContent As String
    Dim expireDate As Date
    Dim today As Date
    Dim fileNumber As Integer
    
    ' Company intranet server file path
    filePath = "\\172.16.232.67\sap\CPO\expire.txt"
    
    ' Read file from network location
    fileNumber = FreeFile()
    Open filePath For Input As fileNumber
    
    ' Read file content
    Line Input #fileNumber, fileContent
    Close fileNumber
    
    ' Parse and validate response
    fileContent = Trim(Replace(Replace(fileContent, vbLf, ""), vbCr, ""))
    
    If Not IsDate(fileContent) Then
        DisplayErrorMessage "Invalid expiration date format in license file."
        CheckExpiration = False
        Exit Function
    End If
    
    ' Check expiration
    expireDate = DateValue(fileContent)
    today = Date
    
    If today > expireDate Then
        DisplayExpirationMessage expireDate, today
        CheckExpiration = False
    Else
        CheckExpiration = True
    End If
    
    Exit Function

ErrorHandler:
    Select Case Err.Number
        Case 53 ' File not found
            DisplayErrorMessage "License file not found at: " & filePath & vbCrLf & "Please ensure the file exists and the path is correct."
        Case 70 ' Permission denied
            DisplayErrorMessage "Permission denied accessing license file." & vbCrLf & "Please check your network permissions."
        Case 75 ' Path/file access error
            DisplayErrorMessage "Path or file access error." & vbCrLf & "Please check network connectivity and file permissions."
        Case Else
            DisplayErrorMessage "Error during license verification: " & Err.Description & " (Error #" & Err.Number & ")"
    End Select
    CheckExpiration = False
End Function

' Display error message
Private Sub DisplayErrorMessage(ByVal message As String)
    Dim fullMessage As String
    fullMessage = message & vbCrLf & vbCrLf & _
                  "Please contact author: Liu Yang" & vbCrLf & _
                  "Email: luckilyliuyang@163.com"
    MsgBox fullMessage, vbCritical, "License Verification Failed"
End Sub

' Display expiration message
Private Sub DisplayExpirationMessage(ByVal expDate As Date, ByVal currDate As Date)
    Dim message As String
    message = "Version expired." & vbCrLf & vbCrLf & _
              "Expiration Date: " & Format(expDate, "yyyy-mm-dd") & vbCrLf & _
              "Current Date: " & Format(currDate, "yyyy-mm-dd") & vbCrLf & vbCrLf & _
              "Please contact author for updated version:" & vbCrLf & _
              "Liu Yang" & vbCrLf & _
              "Email: luckilyliuyang@163.com"
    MsgBox message, vbCritical, "Version Expired"
End Sub

' =============================================
' Worksheet Activate Event with Protection
' Place this in the DATA sheet module
' =============================================
Private Sub Worksheet_Activate()
    ' Execute expiration verification first
    If Not CheckExpiration() Then
        ' If expired, deactivate this sheet
        Application.EnableEvents = False
        If ThisWorkbook.Sheets.Count > 1 Then
            ThisWorkbook.Sheets(1).Activate
        End If
        Application.EnableEvents = True
        Exit Sub
    End If
    
    ' Continue with normal activation if license valid
    Range("A2").Select
End Sub
