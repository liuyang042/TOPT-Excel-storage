Private Sub Workbook_Open()
    ' 启动时立即隐藏Excel窗口，禁用屏幕刷新
    Application.Visible = False
    Application.ScreenUpdating = False
    Application.EnableEvents = False ' 禁用事件避免干扰
    
    Dim githubLink As String
    Dim http As Object
    Dim expireDate As Date
    Dim today As Date
    
    ' 替换为你的GitHub原始链接
    githubLink = "https://raw.githubusercontent.com/liuyang042/ExcelLicense/refs/heads/main/expire.txt"
    
    ' 读取GitHub上的有效期
    On Error Resume Next
    Set http = CreateObject("MSXML2.XMLHTTP.6.0")
    http.Open "GET", githubLink, False
    http.Send
    On Error GoTo 0
    
    ' 检查网络或链接是否有效
    If http Is Nothing Or http.Status <> 200 Then
        MsgBox "Failed to verify version. Please contact author: Liu Yang Email: luckilyliuyang@163.com", vbCritical
        CloseExcelCompletely
        Exit Sub
    End If
    
    ' 解析有效期日期（处理换行和空格）
    On Error Resume Next
    expireDate = DateValue(Trim(Replace(http.ResponseText, vbLf, "")))
    On Error GoTo 0
    
    ' 检查日期格式是否正确
    If expireDate = 0 Then
        MsgBox "Invalid expiration date format. Please contact author.", vbCritical
        CloseExcelCompletely
        Exit Sub
    End If
    
    ' 验证是否过期
    today = Date
    If today > expireDate Then
        MsgBox "Version expired. Please contact author: Liu Yang Email: luckilyliuyang@163.com", vbCritical
        CloseExcelCompletely
    Else
        ' 未过期时，先强制刷新界面，再显示窗口
        ThisWorkbook.Sheets(1).Activate ' 激活首个工作表
        Application.ScreenUpdating = True
        Application.EnableEvents = True
        Application.Visible = True
    End If
End Sub

' 完全关闭Excel程序（确保无残留）
Private Sub CloseExcelCompletely()
    ThisWorkbook.Close SaveChanges:=False
    Application.Quit
    ' 强制终止Excel进程（防止部分环境下Application.Quit失效）
    Dim wsh As Object
    Set wsh = CreateObject("WScript.Shell")
    wsh.Run "taskkill /f /im excel.exe", 0, False
End Sub
