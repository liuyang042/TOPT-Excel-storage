' 有效期检查函数
Function CheckExpiration() As Boolean
    Dim githubLink As String
    Dim http As Object
    Dim expireDate As Date
    Dim today As Date
    
    ' 替换为你的GitHub有效期文件链接
    githubLink = "https://raw.githubusercontent.com/liuyang042/ExcelLicense/refs/heads/main/expire.txt"
    
    ' 读取GitHub上的有效期
    On Error Resume Next
    Set http = CreateObject("MSXML2.XMLHTTP.6.0")
    http.Open "GET", githubLink, False
    http.Send
    On Error GoTo 0
    
    ' 检查网络或链接是否有效
    If http Is Nothing Or http.Status <> 200 Then
        MsgBox "Failed to verify version. Please contact author: Liu Yang Email: luckilyliuyang@163.com", vbCritical
        CheckExpiration = False
        Exit Function
    End If
' 解析有效期日期（处理换行和空格）
    On Error Resume Next
    expireDate = DateValue(Trim(Replace(http.ResponseText, vbLf, "")))
    On Error GoTo 0
    
    ' 检查日期格式是否正确
    If expireDate = 0 Then
        MsgBox "Invalid expiration date format. Please contact author.", vbCritical
        CheckExpiration = False
        Exit Function
    End If
    
    ' 验证是否过期
    today = Date
    If today > expireDate Then
        MsgBox "Version expired. Please contact author: Liu Yang Email: luckilyliuyang@163.com", vbCritical
        CheckExpiration = False
    Else
        CheckExpiration = True
    End If
End Function
